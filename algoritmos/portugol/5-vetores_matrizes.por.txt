algoritmo "ExemploVetorMatriz"
var
    notas: vetor[1..5] de real
    tabela: vetor[1..3, 1..3] de inteiro
    i, j: inteiro
inicio
    // --- TRABALHANDO COM VETOR ---
    escreval("--- Preenchendo o Vetor de Notas ---")
    para i de 1 ate 5 faca
        escreva("Digite a nota ", i, ": ")
        leia(notas[i])
    fimpara
    
    escreval("")
    escreval("Notas digitadas:")
    para i de 1 ate 5 faca
        escreva(notas[i], " ")
    fimpara
    escreval("")
    escreval("")

    // --- TRABALHANDO COM MATRIZ ---
    escreval("--- Preenchendo a Matriz 3x3 ---")
    para i de 1 ate 3 faca
        para j de 1 ate 3 faca
            tabela[i, j] <- i * j
        fimpara
    fimpara
    
    escreval("Matriz gerada:")
    para i de 1 ate 3 faca
        para j de 1 ate 3 faca
            escreva(tabela[i, j], " ")
        fimpara
        escreval("")
    fimpara
fimalgoritmo
